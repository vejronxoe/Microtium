#version 330 core
layout (location = 0) in vec2 aPos;
layout (location = 1) in vec2 aTexCoord;

out vec2 texCoord;

uniform mat4 CrownCamera;
uniform mat4 CrownTransform;
uniform mat4 CrownRotation;

void main()
{
	texCoord = aTexCoord;
	gl_Position = CrownCamera * CrownTransform * CrownRotation *  vec4(aPos.x, aPos.y, 0.0f, 1.0f);
}
