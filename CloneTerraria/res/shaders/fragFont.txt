#version 330 core
out vec4 FragColor;

float median(float r, float g, float b) {
    return max(min(r, g), min(max(r, g), b));
}

in vec2 texCoord;

uniform sampler2D ourTexture;
uniform vec4 uColor;

void main()
{
	vec3 msdf = texture(ourTexture, texCoord).rgb;
	float sd = median(msdf.r, msdf.g, msdf.b);
	float edge = fwidth(sd);
	float alpha = smoothstep(0.5 - edge, 0.5 + edge, sd);
	FragColor = vec4(uColor.rgb, alpha * uColor.a);
}
