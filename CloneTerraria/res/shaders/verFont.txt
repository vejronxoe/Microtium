#version 330 core
layout (location = 0) in vec2 aPos;
layout (location = 1) in vec2 aTexCoord;
layout (location = 2) in vec4 aColor;

out vec2 vTexCoord;
out vec4 vColor;

uniform mat4 camera;
uniform mat4 transform;

void main()
{
	vColor = aColor;
	vTexCoord = aTexCoord;
	gl_Position = camera * transform *vec4(aPos.x, aPos.y, 0.0f, 1.0f);
}
