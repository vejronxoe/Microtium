#version 330 core
layout (location = 0) in vec2 aPos;
layout (location = 1) in vec2 aTexCoord;

out vec2 texCoord;

uniform mat4 HUDCamera;
uniform mat4 HUDTransform;
uniform mat4 HUDScale;
uniform mat4 HUDBasicLocation;

void main()
{
	texCoord = aTexCoord;
	gl_Position =   HUDCamera * HUDTransform * HUDBasicLocation * HUDScale * vec4(aPos.x, aPos.y, 0.0f, 1.0f);
}
