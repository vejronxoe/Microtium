#version 330 core
layout (location = 0) in vec2 aPos;
layout (location = 1) in vec2 aTexCoord;

out vec2 texCoord;

uniform mat4 fontCamera;
uniform mat4 fontTransform;
uniform int fontLetter;
uniform mat4 fontScale;

void main()
{
	texCoord = vec2(fontLetter * 0.1f + aTexCoord.x,aTexCoord.y);
	gl_Position =   fontCamera * fontTransform * fontScale * vec4(aPos.x, aPos.y, 0.0f, 1.0f);
}
