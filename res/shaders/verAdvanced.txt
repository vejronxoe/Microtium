#version 330 core
layout (location = 0) in vec2 aPos;
layout (location = 1) in vec2 aTexCoord;

out vec2 texCoord;

uniform mat4 camera;
uniform mat4 transform;
uniform mat4 rotation;
uniform mat4 scale;
void main()
{
	texCoord = aTexCoord;
	gl_Position = camera * transform * scale * rotation *  vec4(aPos.x, aPos.y, 0.0f, 1.0f);
}
