#version 330 core
layout (location = 0) in vec2 aPos;
layout (location = 1) in vec2 aTexCoord;

out vec2 texCoord;

uniform mat4 animCamera;
uniform mat4 animTransform;
uniform mat4 animScale;
uniform int animNumber;
uniform int animLeangth;

void main()
{
	texCoord = vec2(animNumber * 1.0f/animLeangth + aTexCoord.x,aTexCoord.y);
	gl_Position =   animCamera * animTransform  * animScale * vec4(aPos.x, aPos.y, 0.0f, 1.0f);
}
